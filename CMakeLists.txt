cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(JGJGsTransmogSystem VERSION 0.1.0 LANGUAGES CXX)
option(JGJG_BUILD_SKSE "Build the Windows SKSE catalogue prototype" OFF)
add_library(jgjg_core INTERFACE)
target_include_directories(jgjg_core INTERFACE include)
target_compile_features(jgjg_core INTERFACE cxx_std_20)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE jgjg_core)
enable_testing()
add_test(NAME core COMMAND core_tests)
if(JGJG_BUILD_SKSE)
    if(NOT WIN32 OR NOT MSVC)
        message(FATAL_ERROR "SKSE build requires Windows x64 and MSVC")
    endif()
    include(FetchContent)
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(CommonLibSSE GIT_REPOSITORY https://github.com/CharmedBaryon/CommonLibSSE-NG.git GIT_TAG b93280e832f263dbef44e44cbe2936622a02f91a)
    FetchContent_MakeAvailable(CommonLibSSE)
    # SafetyHook supplies a relocated trampoline for the engine's armor-model
    # application function. Pin the release so CI and user builds agree.
    set(SAFETYHOOK_FETCH_ZYDIS ON CACHE BOOL "" FORCE)
    FetchContent_Declare(SafetyHook GIT_REPOSITORY https://github.com/cursey/safetyhook.git GIT_TAG v0.7.0)
    FetchContent_MakeAvailable(SafetyHook)
    target_compile_options(CommonLibSSE PRIVATE /MP3)
    include(${commonlibsse_SOURCE_DIR}/cmake/CommonLibSSE.cmake)
    add_commonlibsse_plugin(JGJGsTransmogSystem NAME "JGJGs Transmog System" AUTHOR "JGJG" USE_ADDRESS_LIBRARY SOURCES src/Plugin.cpp src/ArmorAppearance.cpp src/WeaponAppearance.cpp)
    target_link_libraries(JGJGsTransmogSystem PRIVATE jgjg_core)
    target_link_libraries(JGJGsTransmogSystem PRIVATE safetyhook)
    target_compile_features(JGJGsTransmogSystem PRIVATE cxx_std_23)
    target_precompile_headers(JGJGsTransmogSystem PRIVATE src/CommonLibLiterals.hpp <RE/Skyrim.h> <SKSE/SKSE.h> <safetyhook.hpp>)
endif()
